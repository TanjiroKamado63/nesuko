C. CALCULATE THE SUM OF TWO NUMBERS USING A FUNCTION

File: src/App.js

function App() {
    const num1 = 25;
    const num2 = 15;

    function calculateSum(a, b) {
        return a + b;
    }

    return (
        <div>
            <h2>Sum Calculation</h2>
            <p>Number 1: {num1}</p>
            <p>Number 2: {num2}</p>
            <p>Sum: {calculateSum(num1, num2)}</p>
        </div>
    );
}

export default App;

F. DISPLAY CURRENT DATE AND TIME DYNAMICALLY

File: src/DateTime.jsx

function DateTime() {
    const currentDate = new Date();

    return (
        <div>
            <h2>Current Date and Time</h2>
            <p>Date: {currentDate.toLocaleDateString()}</p>
            <p>Time: {currentDate.toLocaleTimeString()}</p>
        </div>
    );
}

export default DateTime;


File: src/App.js

import DateTime from "./DateTime";

function App() {
    return (
        <div>
            <h1>Date and Time Example</h1>
            <DateTime />
        </div>
    );
}

export default App;
