D. PROGRAM USING ARRAY METHODS (map, filter, reduce)

let numbers = [10, 20, 30, 40, 50];

// map()
let doubled = numbers.map(num => num * 2);

// filter()
let greaterThan25 = numbers.filter(num => num > 25);

// reduce()
let total = numbers.reduce((sum, num) => sum + num, 0);

console.log("Original:", numbers);
console.log("Doubled:", doubled);
console.log("Greater than 25:", greaterThan25);
console.log("Sum:", total);

F. COMBINE FUNCTIONS, ARRAYS AND OBJECTS

// Array of student objects
const students = [
    { name: "Alice", marks: 85 },
    { name: "Bob", marks: 72 },
    { name: "Charlie", marks: 90 },
    { name: "David", marks: 65 }
];

// Function to calculate average marks
const averageMarks = (students) => {
    const total = students.reduce((sum, student) => sum + student.marks, 0);
    return total / students.length;
};

// Function to find passed students
const passedStudents = (students) => {
    return students.filter(student => student.marks >= 75);
};

console.log("Average Marks:", averageMarks(students));

console.log("Passed Students:");
passedStudents(students).forEach(student => {
    console.log(`${student.name} - ${student.marks}`);
});
