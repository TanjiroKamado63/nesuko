PRACTICAL NO. 8 – API CREATION AND INTEGRATION

AIM
a. Create a simple REST API using Express.
b. Implement a GET API to retrieve data.
c. Fetch data from the API in a React application.
d. Send data from React to the server using a POST request.
e. Store data in MongoDB.
f. Display stored data in React page.

INSTALL REQUIRED PACKAGES

npm install express mongoose cors


HOW TO RUN PRACTICALS 8, 9 AND 10

1. Open one terminal and go to the directory containing server.js (the practical PDF describes this as the src directory).
2. Run: node server.js
3. Keep the backend server running.
4. In another terminal, create/run the React application using npm start.


FILE: server.js

const express = require("express");
const mongoose = require("mongoose");
const cors = require("cors");

const app = express();

app.use(cors());
app.use(express.json());

// Connect to MongoDB
mongoose
    .connect("mongodb://127.0.0.1:27017/studentDB")
    .then(() => console.log("MongoDB Connected"))
    .catch((err) => console.log(err));

// Schema
const studentSchema = new mongoose.Schema({
    name: String,
    age: Number
});

// Model
const Student = mongoose.model("Student", studentSchema);

// GET API - retrieve data
app.get("/students", async (req, res) => {
    const students = await Student.find();
    res.json(students);
});

// POST API - receive and store data
app.post("/students", async (req, res) => {
    const student = new Student({
        name: req.body.name,
        age: req.body.age
    });

    await student.save();

    res.json({
        message: "Student saved successfully",
        student: student
    });
});

// Start server
app.listen(5000, () => {
    console.log("Server running on port 5000");
});


FILE: App.jsx

import { useEffect, useState } from "react";

function App() {
    const [students, setStudents] = useState([]);
    const [name, setName] = useState("");
    const [age, setAge] = useState("");

    // GET data from API
    const getStudents = async () => {
        const response = await fetch("http://localhost:5000/students");
        const data = await response.json();
        setStudents(data);
    };

    // POST data to API
    const addStudent = async (e) => {
        e.preventDefault();

        await fetch("http://localhost:5000/students", {
            method: "POST",
            headers: {
                "Content-Type": "application/json"
            },
            body: JSON.stringify({
                name: name,
                age: Number(age)
            })
        });

        setName("");
        setAge("");

        // Refresh displayed data
        getStudents();
    };

    // Get students when page loads
    useEffect(() => {
        getStudents();
    }, []);

    return (
        <div>
            <h1>Student Management</h1>

            <form onSubmit={addStudent}>
                <input
                    type="text"
                    placeholder="Enter name"
                    value={name}
                    onChange={(e) => setName(e.target.value)}
                />

                <input
                    type="number"
                    placeholder="Enter age"
                    value={age}
                    onChange={(e) => setAge(e.target.value)}
                />

                <button type="submit">Add Student</button>
            </form>

            <h2>Students</h2>

            {students.map((student) => (
                <p key={student._id}>
                    {student.name} - {student.age} years
                </p>
            ))}
        </div>
    );
}

export default App;


OUTPUT
The React page displays a Student Management form with name and age fields and an Add Student button. Stored students are displayed below the form. The API uses MongoDB database studentDB and runs on port 5000.
